Tres trampas clásicas de UVa: (1) la fatiga es un valor fijo Δ = U · F / 100, NO un porcentaje del ascenso del día anterior; (2) el ascenso nunca puede ser negativo (max(0.0, climb)); (3) se verifica éxito (> H) inmediatamente después de subir de día y fracaso (< 0) después de resbalar de noche (altura = 0 aún no falla, altura = H aún no sale).

#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    double H, U, D, F;
    while (cin >> H >> U >> D >> F && H != 0) {
        double fatigue = U * (F / 100.0);
        double height = 0.0;
        double climb = U;
        int day = 0;

        while (true) {
            day++;
            if (climb > 0.0) height += climb;
            if (height > H) {
                cout << "success on day " << day << "\n";
                break;
            }
            height -= D;
            if (height < 0.0) {
                cout << "failure on day " << day << "\n";
                break;
            }
            climb = max(0.0, climb - fatigue);
        }
    }
    return 0;
}

Si una capacidad máxima C es suficiente para usar ≤ m contenedores, cualquier capacidad mayor también lo es (monotonía). Por tanto, hacemos búsqueda binaria en [max(c), sum(c)] y verificamos factibilidad en O(n) llenando cada contenedor lo máximo posible antes de abrir uno nuevo.

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

bool canFill(const vector<long long>& c, int m, long long cap) {
    int used = 1;
    long long cur = 0;
    for (long long v : c) {
        if (v > cap) return false;
        if (cur + v > cap) {
            used++;
            cur = v;
        } else {
            cur += v;
        }
    }
    return used <= m;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;

        }
        cout << lo << "\n";
    }
    return 0;
}

Mantener dos heaps balanceados: un max-heap lo con la mitad menor y un min-heap hi con la mitad mayor, preservando |lo| == |hi| o |lo| == |hi| + 1. Cada inserción toma O(log N) y consultar la mediana es O(1). Usar enteros de 64 bits al sumar los dos elementos centrales para evitar overflow.

#include <iostream>
#include <queue>
#include <vector>
#include <functional>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    priority_queue<long long> lo; // max-heap (mitad inferior)
    priority_queue<long long, vector<long long>, greater<long long>> hi; // min-heap

    long long x;
    while (cin >> x) {
        if (lo.empty() || x <= lo.top()) lo.push(x);
        else hi.push(x);

        // Rebalancear tamaños: |lo| == |hi| o |lo| == |hi| + 1
        if (lo.size() > hi.size() + 1) {
            hi.push(lo.top());
            lo.pop();
        } else if (hi.size() > lo.size()) {
            lo.push(hi.top());
            hi.pop();
        }

        if (lo.size() > hi.size()) {
            cout << lo.top() << "\n";
        } else {
            cout << (lo.top() + hi.top()) / 2 << "\n";
        }
    }
    return 0;
}

Asignar un entero secuencial a cada nombre nuevo mediante un diccionario (unordered_map) y mantener un DSU con arreglo sz[]. Al unir dos representantes distintos ra y rb, sumar sz[ra] += sz[rb]; si ya estaban en el mismo conjunto, no duplicar el tamaño y devolver sz[ra] directamente.

#include <iostream>
#include <vector>
#include <string>
#include <unordered_map>
#include <utility>
using namespace std;

struct DSU {
    vector<int> p, sz;
    int addNode() {
        int id = (int)p.size();
        p.push_back(id);
        sz.push_back(1);
        return id;
    }
    int find(int x) {
        return p[x] == x ? x : (p[x] = find(p[x]));
    }
    int unite(int a, int b) {
        a = find(a); b = find(b);
        if (a != b) {
            if (sz[a] < sz[b]) swap(a, b);
            p[b] = a;
            sz[a] += sz[b];
        }
        return sz[a];
    }
};

El estado memo[u][mask] guarda el costo mínimo de completar el tour estando en el punto u cuando el conjunto de puntos ya visitados está codificado en mask. Cuando mask == (1 << N) - 1, solo resta volver al origen dist(u, 0).

#include <iostream>
#include <vector>
#include <cmath>
#include <algorithm>
using namespace std;

int N;
int x[12], y[12], distM[12][12];
int memo[12][1 << 12];

int tsp(int pos, int mask) {
    if (mask == (1 << N) - 1) return distM[pos][0];
    int& ans = memo[pos][mask];
    if (ans != -1) return ans;
    ans = 1e9;
    for (int nxt = 0; nxt < N; nxt++) {
        if (!(mask & (1 << nxt))) {
            ans = min(ans, distM[pos][nxt] + tsp(nxt, mask | (1 << nxt)));
        }
    }
    return ans;
}

struct DSU {
    vector<int> p, r;
    DSU(int n) : p(n), r(n, 0) { iota(p.begin(), p.end(), 0); }
    int find(int x) { return p[x] == x ? x : (p[x] = find(p[x])); }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (r[a] < r[b]) swap(a, b);
        p[b] = a;
        if (r[a] == r[b]) r[a]++;
        return true;
    }

Con el algoritmo de Floyd: (1) avanzamos tortuga t = f(x₀) y liebre h = f(f(x₀)) hasta que coincidan dentro del ciclo; (2) reiniciamos h = x₀ y avanzamos ambos de a 1 paso para hallar el inicio μ del ciclo; (3) fijamos t y avanzamos h = f(t) contando pasos hasta volver a t, obteniendo exactamente λ en O(1) memoria.

#include <iostream>

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    long long Z, I, M, L;
    int tc = 1;
    while (cin >> Z >> I >> M >> L && (Z || I || M || L)) {
        auto f = [&](long long x) { return (Z * x + I) % M; };

        // Fase 1: encontrar punto de encuentro dentro del ciclo
        long long t = f(L), h = f(f(L));
        while (t != h) {
            t = f(t);
            h = f(f(h));
        }
        // Fase 2: (opcional aquí, pero completo) inicio mu del ciclo
        h = L;
        while (t != h) {
            t = f(t);
            h = f(h);
        }
        // Fase 3: medir longitud lambda del ciclo
        long long lambda = 1;
        h = f(t);
        while (t != h) {
            h = f(h);
            lambda++;
        }
        cout << "Case " << tc++ << ": " << lambda << "\n";
    }
    return 0;

Al calcular la función prefijo pi de KMP en O(N), pi[n - 1] es el borde propio más largo (prefijo que también es sufijo). Entonces p = n - pi[n - 1] es el candidato a periodo mínimo: si n % p == 0, la cadena es exactamente periódica con k = n / p repeticiones; en caso contrario no se repite completamente y k = 1.

        if (s == ".") break;

        vector<int> pi(n, 0);
        for (int i = 1, j = 0; i < n; i++) {
            while (j > 0 && s[i] != s[j]) j = pi[j - 1];
            if (s[i] == s[j]) j++;
            pi[i] = j;
        }
        int period = n - pi[n - 1];
        if (n % period == 0) cout << (n / period) << "\n";
        else cout << 1 << "\n";
    }
    return 0;
}

struct Pt {
    long long x, y;

        return x != o.x ? x < o.x : y < o.y;

long long cross(const Pt& O, const Pt& A, const Pt& B) {
    return (A.x - O.x) * (B.y - O.y) - (A.y - O.y) * (B.x - O.x);
}

long long area2(const vector<Pt>& P) {
    long long s = 0;
    int n = (int)P.size();
    for (int i = 0; i < n; i++) {
        int j = (i + 1) % n;
        s += P[i].x * P[j].y - P[j].x * P[i].y;
    }
    return llabs(s);
}

}

public class Main {
    static class Pt implements Comparable<Pt> {
        long x, y;
        Pt(long x, long y) { this.x = x; this.y = y; }
        public int compareTo(Pt o) {
            return x != o.x ? Long.compare(x, o.x) : Long.compare(y, o.y);
        }
    }

    static long cross(Pt O, Pt A, Pt B) {
        return (A.x - O.x) * (B.y - O.y) - (A.y - O.y) * (B.x - O.x);
    }

    static long area2(Pt[] P, int n) {
        long s = 0;
        for (int i = 0; i < n; i++) {
            int j = (i + 1) % n;
            s += P[i].x * P[j].y - P[j].x * P[i].y;
        }
        return Math.abs(s);
    }

    static Pt[] convexHull(Pt[] orig) {
        int n = orig.length, k = 0;
        if (n <= 1) return orig;
        Pt[] P = orig.clone();
        Arrays.sort(P);
        Pt[] H = new Pt[2 * n];
        for (int i = 0; i < n; i++) {
            while (k >= 2 && cross(H[k - 2], H[k - 1], P[i]) <= 0) k--;
            H[k++] = P[i];
        }
        for (int i = n - 2, t = k + 1; i >= 0; i--) {
            while (k >= t && cross(H[k - 2], H[k - 1], P[i]) <= 0) k--;
            H[k++] = P[i];
        }
        return Arrays.copyOf(H, k - 1);
    }

    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static StringTokenizer st;

    static String next() throws IOException {
        while (st == null || !st.hasMoreTokens()) {
            String line = br.readLine();
            if (line == null) return null;
            st = new StringTokenizer(line);
        }
        return st.nextToken();
    }

    public static void main(String[] args) throws Exception {
        StringBuilder out = new StringBuilder();
        int tc = 1;
        String tok;
        while ((tok = next()) != null) {
            int n = Integer.parseInt(tok);
            if (n == 0) break;
            Pt[] poly = new Pt[n];
            for (int i = 0; i < n; i++) {
                poly[i] = new Pt(Long.parseLong(next()), Long.parseLong(next()));
            }
            long aPoly = area2(poly, n);
            Pt[] hull = convexHull(poly);
            long aHull = area2(hull, hull.length);
            double wasted = (1.0 - (double) aPoly / (double) aHull) * 100.0;
            out.append("Tile #").append(tc++).append('\n');
            out.append(String.format(Locale.US, "Wasted Space = %.2f %%\n\n", wasted));
        }
        System.out.print(out);
    }

import sys

def cross(O, A, B):
    return (A[0] - O[0]) * (B[1] - O[1]) - (A[1] - O[1]) * (B[0] - O[0])

def area2(P):
    n = len(P)
    s = 0
    for i in range(n):
        x1, y1 = P[i]
        x2, y2 = P[(i + 1) % n]
        s += x1 * y2 - x2 * y1
    return abs(s)

def convex_hull(pts):
    pts = sorted(pts)
    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]
